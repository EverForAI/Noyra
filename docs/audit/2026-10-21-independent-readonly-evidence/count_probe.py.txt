
import json,sys,time,os,tempfile,shutil
from pathlib import Path
sys.path.insert(0,str(Path(sys.argv[1])/'src'))
from noyra.core.at_rest import AtRestGuard,AtRestConfig,VolumeEncryptionStatus,_private_permission_error
from noyra.core import Database,IdentityStore
from noyra.core.types import content_hash
from noyra.core.admission import RuntimeAdmissionGate,bind_lease
from noyra.capability import CapabilityStore,CapabilityGrant,ToolRunner
import noyra.capability.tools as toolmod

out={"source":sys.argv[1]}
root=Path(tempfile.mkdtemp(prefix='noyra-audit-probe-',dir='/var/tmp'));os.chmod(root,0o700)
os.umask(0o077)
class Key:
 def metadata(self):return {"probe":True}
try:
 data=root/'private';data.mkdir(mode=0o700)
 for i in range(100002):
  fd=os.open(data/f'f{i:06}',os.O_CREAT|os.O_WRONLY,0o600);os.close(fd)
 config=AtRestConfig(mode='required',data_root=data)
 guard=AtRestGuard(config)
 def volume():
  guard._volume=VolumeEncryptionStatus(True,'audit-isolated','probe')
  return guard._volume
 guard._require_volume=volume;guard._require_keyring=lambda:Key();guard._prepared=True
 t=time.monotonic();strict=guard.health();duration=time.monotonic()-t
 health={"entries":100003,"strict_ready":strict['ready'],"strict_seconds":duration}
 if hasattr(guard,'request_health'):health['count_only_scan_error']=_private_permission_error(data,deadline=time.monotonic()+600)
 if hasattr(guard,'request_health'):
  guard.request_health();time.sleep(3)
  health['request_samples']=[]
  for _ in range(3):
   h=guard.request_health();health['request_samples'].append({k:h.get(k) for k in ('ready','reason')});time.sleep(1)
  guard.close()
 else:health['request_health']='not present; baseline service uses strict health()'
 out['large_tree']=health

 db=Database(root/'audit.sqlite3');IdentityStore(db).ensure('audit',content_hash('audit'))
 target=root/'test.txt';target.write_text('allowed',encoding='utf-8')
 CapabilityStore(db).grant('audit',CapabilityGrant(capability_type='filesystem_read',scope={'root':str(root)},issuer='owner',rate_limit_per_hour=10,side_effect=False),actor='operator')
 gate=RuntimeAdmissionGate('audit');original=toolmod.read_bounded
 def race(*a,**kw):
  result=original(*a,**kw);gate.invalidate();return result
 toolmod.read_bounded=race
 result={}
 with gate.operation('independent-read-probe') as lease,bind_lease(lease):
  try:result['return']=str(ToolRunner(db).read_text('audit',target,idempotency_key='race'))
  except BaseException as e:result['error']=type(e).__name__
 with db.connection() as conn:result['actions']=[dict(row) for row in conn.execute('SELECT action_type,status FROM actions')]
 out['read_lease_invalidated']=result
finally:
 shutil.rmtree(root)
print(json.dumps(out,ensure_ascii=False,indent=2))
